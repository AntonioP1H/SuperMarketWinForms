CREATE PROCEDURE sp_SearchProducts @search VARCHAR(100) AS
BEGIN
    SELECT p.pcode, p.barcode, p.pdesc, b.brand, c.category, p.price, p.reorder 
    FROM tbProduct AS p 
    INNER JOIN tbBrand AS b ON b.id = p.bid 
    INNER JOIN tbCategory AS c ON c.id = p.cid 
    WHERE CONCAT(p.pdesc, b.brand, c.category) LIKE @search;
END
GO

CREATE PROCEDURE sp_GetCartItems
    @transno VARCHAR(50)
AS
BEGIN
    SELECT c.id, p.pcode, p.pdesc, c.price, c.qty, c.disc, c.total, c.cashier 
    FROM tbCart c 
    INNER JOIN tbProduct p ON c.pcode = p.pcode 
    WHERE c.transno = @transno AND c.status = 'Pending';
END
GO

CREATE PROCEDURE sp_AddToCart
    @transno VARCHAR(50),
    @pcode VARCHAR(50),
    @price DECIMAL(18,2),
    @qty INT,
    @cashier VARCHAR(50)
AS
BEGIN
    IF EXISTS (SELECT * FROM tbCart WHERE transno = @transno AND pcode = @pcode AND status = 'Pending')
        UPDATE tbCart SET qty = qty + @qty WHERE transno = @transno AND pcode = @pcode AND status = 'Pending';
    ELSE
        INSERT INTO tbCart(transno, pcode, price, qty, sdate, cashier, status)
        VALUES (@transno, @pcode, @price, @qty, GETDATE(), @cashier, 'Pending');
END
GO

CREATE PROCEDURE sp_SettlePayment
    @transno VARCHAR(50)
AS
BEGIN
    UPDATE p 
    SET p.qty = p.qty - c.qty 
    FROM tbProduct p 
    INNER JOIN tbCart c ON p.pcode = c.pcode 
    WHERE c.transno = @transno AND c.status = 'Pending';

    UPDATE tbCart SET status = 'Sold' WHERE transno = @transno AND status = 'Pending';
END
GO

CREATE PROCEDURE sp_GetDailySales
    @datefrom DATE,
    @dateto DATE,
    @cashier VARCHAR(50)
AS
BEGIN
    IF @cashier = 'All Cashiers'
        SELECT c.id, c.transno, p.pcode, p.pdesc, c.price, c.qty, c.disc, c.total, c.sdate, c.cashier 
        FROM tbCart c 
        INNER JOIN tbProduct p ON c.pcode = p.pcode 
        WHERE c.status = 'Sold' AND CAST(c.sdate AS DATE) BETWEEN @datefrom AND @dateto;
    ELSE
        SELECT c.id, c.transno, p.pcode, p.pdesc, c.price, c.qty, c.disc, c.total, c.sdate, c.cashier 
        FROM tbCart c 
        INNER JOIN tbProduct p ON c.pcode = p.pcode 
        WHERE c.status = 'Sold' AND c.cashier = @cashier AND CAST(c.sdate AS DATE) BETWEEN @datefrom AND @dateto;
END
GO
